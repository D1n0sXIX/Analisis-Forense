================================================================
  1. CAPTURA DE TRAFICO
================================================================

# Listar interfaces disponibles
tshark -D

# Capturar en una interfaz
tshark -i eth0

# Capturar y guardar en fichero PCAP
tshark -i eth0 -w captura.pcap

# Capturar N paquetes y parar
tshark -i eth0 -c 100

# Leer un fichero PCAP existente
tshark -r captura.pcap


================================================================
  2. FILTROS DE CAPTURA (mientras captura)
================================================================

# Filtrar por protocolo
tshark -i eth0 -f "tcp"
tshark -i eth0 -f "udp"
tshark -i eth0 -f "icmp"

# Filtrar por puerto
tshark -i eth0 -f "port 80"
tshark -i eth0 -f "port 443"
tshark -i eth0 -f "port 21 or port 22"

# Filtrar por host
tshark -i eth0 -f "host 192.168.1.1"
tshark -i eth0 -f "src host 192.168.1.1"
tshark -i eth0 -f "dst host 192.168.1.1"


================================================================
  3. FILTROS DE VISUALIZACION (sobre PCAP ya capturado)
================================================================

# Filtrar por protocolo
tshark -r captura.pcap -Y "http"
tshark -r captura.pcap -Y "dns"
tshark -r captura.pcap -Y "ftp"
tshark -r captura.pcap -Y "ssh"
tshark -r captura.pcap -Y "tcp"

# Filtrar por IP
tshark -r captura.pcap -Y "ip.addr == 192.168.1.1"
tshark -r captura.pcap -Y "ip.src == 192.168.1.1"
tshark -r captura.pcap -Y "ip.dst == 192.168.1.1"

# Filtrar por puerto
tshark -r captura.pcap -Y "tcp.port == 80"
tshark -r captura.pcap -Y "tcp.port == 443"
tshark -r captura.pcap -Y "udp.port == 53"

# Filtrar por flags TCP
tshark -r captura.pcap -Y "tcp.flags.syn == 1"
tshark -r captura.pcap -Y "tcp.flags.reset == 1"
tshark -r captura.pcap -Y "tcp.flags.syn == 1 and tcp.flags.ack == 0"

# Filtrar solo queries DNS (no respuestas)
tshark -r captura.pcap -Y "dns.flags.response == 0"

# Filtrar solo respuestas DNS
tshark -r captura.pcap -Y "dns.flags.response == 1"

# Filtrar HTTP por metodo
tshark -r captura.pcap -Y "http.request.method == GET"
tshark -r captura.pcap -Y "http.request.method == POST"

# Filtrar paquetes con datos (longitud > 0)
tshark -r captura.pcap -Y "data.len > 0"
tshark -r captura.pcap -Y "tcp.len > 0"


================================================================
  4. EXTRAER CAMPOS ESPECIFICOS (-T fields -e campo)
================================================================

# Ver IPs origen y destino
tshark -r captura.pcap -T fields -e ip.src -e ip.dst

# Ver IP, puerto origen y puerto destino
tshark -r captura.pcap -T fields -e ip.src -e tcp.srcport -e ip.dst -e tcp.dstport

# Ver solo dominios de queries DNS
tshark -r captura.pcap -Y "dns.flags.response == 0" \
  -T fields -e dns.qry.name

# Ver metodo HTTP y URI
tshark -r captura.pcap -Y "http.request" \
  -T fields -e http.request.method -e http.request.uri

# Ver codigo de respuesta HTTP y content-type
tshark -r captura.pcap -Y "http.response" \
  -T fields -e http.response.code -e http.content_type

# Ver credenciales FTP
tshark -r captura.pcap -Y "ftp" \
  -T fields -e ftp.request.command -e ftp.request.arg


================================================================
  5. ESTADISTICAS
================================================================

# Resumen de protocolos
tshark -r captura.pcap -q -z io,phs

# Conversaciones TCP
tshark -r captura.pcap -q -z conv,tcp

# Conversaciones UDP
tshark -r captura.pcap -q -z conv,udp

# Estadisticas HTTP
tshark -r captura.pcap -q -z http,tree

# Estadisticas DNS
tshark -r captura.pcap -q -z dns,tree

# IPs que mas trafico generan
tshark -r captura.pcap -q -z endpoints,ip


================================================================
  6. EXTRAER OBJETOS / FICHEROS
================================================================

# Extraer objetos HTTP (ficheros descargados)
tshark -r captura.pcap --export-objects http,/tmp/objetos/

# Extraer objetos FTP
tshark -r captura.pcap --export-objects ftp-data,/tmp/objetos/

# Extraer objetos SMB
tshark -r captura.pcap --export-objects smb,/tmp/objetos/


================================================================
  7. FLUJO COMPLETO DE UN STREAM TCP
================================================================

# Ver el contenido de un stream especifico (equivale a Follow TCP Stream)
tshark -r captura.pcap -q -z follow,tcp,ascii,0
tshark -r captura.pcap -q -z follow,tcp,ascii,1
# (cambiar el numero por el stream index)

# Follow en formato hex
tshark -r captura.pcap -q -z follow,tcp,hex,0

# Follow UDP (ej. DNS tunneling)
tshark -r captura.pcap -q -z follow,udp,ascii,0


================================================================
  8. CASOS PRACTICOS VISTOS EN CLASE
================================================================

--- BEACON / REVERSE SHELL ---
# Ver conversaciones largas y sostenidas
tshark -r captura.pcap -q -z conv,tcp
# Buscar streams con muchos paquetes y larga duracion
# Luego: follow,tcp,ascii,<stream_index>

--- DESCARGA DE FICHERO POR HTTP ---
# Ver peticiones GET
tshark -r captura.pcap -Y "http.request" \
  -T fields -e http.request.method -e http.request.uri
# Extraer fichero
tshark -r captura.pcap --export-objects http,/tmp/objetos/

--- DNS TUNNELING ---
# Detectar: muchas queries DNS con subdominios raros
tshark -r captura.pcap -q -z conv,udp
# Ver las queries
tshark -r captura.pcap -Y "dns.flags.response == 0" \
  -T fields -e dns.qry.name
# Reconstruir datos exfiltrados (si estan en hex):
tshark -r captura.pcap -Y "dns.flags.response == 0" \
  -T fields -e dns.qry.name 2>/dev/null \
  | sed 's/\.dominio\.com//' \
  | tr -d '\n' \
  | xxd -r -p \
  | grep -a "flag"

--- PORT SCAN ---
# Ver conexiones de un solo origen a muchos puertos distintos
tshark -r captura.pcap -q -z conv,tcp
# Caracteristica: muchas conexiones de 2 paquetes (SYN + RST/SYN-ACK)

--- FTP ---
# Ver comandos FTP
tshark -r captura.pcap -Y "ftp" \
  -T fields -e ftp.request.command -e ftp.request.arg
# Extraer ficheros transferidos
tshark -r captura.pcap --export-objects ftp-data,/tmp/ftp/


================================================================
  9. OPCIONES UTILES
================================================================

-r fichero.pcap     Leer fichero PCAP
-w fichero.pcap     Escribir captura a fichero
-i eth0             Interfaz de captura
-Y "filtro"         Filtro de visualizacion (display filter)
-f "filtro"         Filtro de captura (capture filter)
-T fields           Modo de salida por campos
-e campo            Campo a extraer (con -T fields)
-q                  Modo silencioso (solo muestra el resultado)
-z estadistica      Estadisticas
-c N                Capturar solo N paquetes
-V                  Verbose (detalle completo de cada paquete)
2>/dev/null         Suprimir errores/warnings
